# OKF 0.2 Full Adoption Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make every Memoria vault a conformant OKF v0.2 bundle — trust, provenance, and reserved-file structure included — while the alpha's empty install base makes format changes free.

**Architecture:** OKF fields are projections of the SQLite trust plane, never sources of truth: the trusted writer stamps `generated` at staging and `verified` at promotion, derives `sources` from derivation inputs, and the reserved files (`index.md`, `log.md`) are tracked projections. Frontmatter validation stays closed — every new field enters the type-schema rosters. `status`/`stale_after` are deliberate omissions (absent ⇒ `stable` per spec; staging is Memoria's draft plane; adding either later is purely additive, so omission has zero migration cost). The Attested Computation *type schema* is not added (no instances exist; adding it ripples into the concept-type DB CHECK parity) — the spec's unknown-type tolerance covers us.

**Tech Stack:** Python 3.12, PyYAML, SQLite (existing runtime); no new dependencies.

**Spec:** OKF v0.2 — <https://github.com/GoogleCloudPlatform/knowledge-catalog/blob/main/okf/SPEC.md>. Conformance bar (§11): every non-reserved `.md` in the bundle tree has parseable YAML frontmatter with non-empty `type`; reserved `index.md`/`log.md` follow §8/§9 when present. v0.1→v0.2 supersessions (§13.1): `timestamp` → `generated.{by,at}`; body `# Citations` → frontmatter `sources`.

## Global Constraints

- Correctness command: `python scripts/verify` — must pass before any PR; `main` requires PR + `verify` + `gitleaks` checks; merge by squash.
- Work in a worktree: `git worktree add .claude/worktrees/okf-02 -b wip/okf-02 origin/main`, then `EnterWorktree(path: ".claude/worktrees/okf-02")`. Stage explicit paths, never `git add -A`.
- Frontmatter validation is **closed** (`validate_frontmatter` in `src/memoria_vault/runtime/subsystems/lib/schema.py`): any new frontmatter key must be declared in every type yaml that may carry it.
- `RETIRED_FRONTMATTER_FIELDS` (`src/memoria_vault/runtime/vaultio.py:19-34`) bans `status`, `check_status`, `created`, `updated`, `verdict` from frontmatter. Do NOT un-retire anything.
- The checked-terminology gate (`scripts/checks/checked_terminology_gate.py`) fails any docs/src text placing `checked` within 100 chars of `verified`/`trusted`/`approved`. Docs tasks below phrase around this — keep the two vocabularies in separate sentences at distance, exactly as written in the task bodies.
- Trust order when confused: schema → tests → code → docs.

```python
def okf_actor(
    actor: str,
    *,
    agent_identity: str = "",
    operation_id: str = "",
    machine_authored: bool = False,
) -> str:
    """Render provenance in the OKF v0.2 actor grammar (spec §7).

    Authorship follows `machine_authored`, mirroring the trusted writer's
    authority/authorship split: a PI-authority door relaying a machine body
    is a machine author.
    """
    if actor == "pi" and not machine_authored:
        return "human:pi"
    if actor in {"operation", "integrity"}:
        return f"process:{operation_id or actor}"
    identity = agent_identity or "agent"
    return identity if "/" in identity else f"{identity}/unversioned"


def okf_verified_actor(actor: str, *, operation_id: str = "") -> str:
    """Actor for a `verified` event: promotion is PI judgment, so PI
    authority verifies as `human:pi` regardless of body authorship."""
    if actor == "pi":
        return "human:pi"
    return f"process:{operation_id or actor}"
```

def _sources_from_inputs(input_rows: list[dict[str, Any]]) -> list[dict[str, str]]:
    """Project derivation inputs into OKF v0.2 `sources` entries (spec §5.1)."""
    out: list[dict[str, str]] = []
    for row in input_rows:
        rid = str(row.get("id") or "")
        if not rid:
            continue
        slug = rid.removesuffix(".md").replace("/", "-")
        out.append({"id": slug, "resource": f"/{rid}"})
    return out

_LOG_EVENT_LABELS = {
    "derived": "Derivation",
    "check-fired": "Check",
    "disposition": "Disposition",
}


def _vault_log(vault: Path) -> str:
    """Render the OKF v0.2 reserved log.md (spec §9) from the journal."""
    from memoria_vault.runtime import state

    with state.connect(vault) as conn:
        rows = conn.execute(
            "SELECT timestamp, event_type, payload_json FROM event_log ORDER BY event_id DESC"
        ).fetchall()
    by_date: dict[str, list[str]] = {}
    for row in rows:
        day = str(row["timestamp"])[:10]
        payload = json.loads(row["payload_json"])
        label = _LOG_EVENT_LABELS.get(str(row["event_type"]))
        if label is None:
            continue
        subject = str(
            payload.get("output_id") or payload.get("target_id") or payload.get("item_id") or ""
        )
        if not subject:
            continue
        by_date.setdefault(day, []).append(f"* **{label}**: {subject}.")
    sections = "\n\n".join(
        f"## {day}\n" + "\n".join(entries) for day, entries in by_date.items()
    )
    return f"# Vault update log\n\n{sections}\n" if sections else "# Vault update log\n"
```

def validate_okf_core_workspace(root: Path, schemas_dir: Path | None = None) -> list[str]:
    """OKF v0.2 core conformance (spec §11) over the whole bundle tree."""
    from memoria_vault.runtime.vaultio import DEFAULT_SKIP_DIRS

    root = Path(root)
    folders = load_folders(schemas_dir)
    errors: list[str] = []
    for bundle in bundle_roots(folders):
        if not (root / bundle).is_dir():
            errors.append(f"missing bundle root: {bundle}")
    reserved = {"index.md", "log.md"}
    for path in sorted(root.rglob("*.md")):
        if any(part in DEFAULT_SKIP_DIRS for part in path.parts):
            continue
        rel = path.relative_to(root).as_posix()
        if path.name in reserved:
            errors.extend(_reserved_file_errors(path, rel, is_bundle_root=(rel == path.name)))
            continue
        if path.name == "SCHEMA.md":
            continue
        fm, _body, fm_errors = _markdown_frontmatter(path)
        errors.extend(f"{rel}: {err}" for err in fm_errors)
        if fm_errors:
            continue
        if fm.get("type") in (None, ""):
            errors.append(f"{rel}: missing required field: type")
        errors.extend(f"{rel}: {error}" for error in universal_concept_frontmatter_errors(fm, rel))
    return errors


def _reserved_file_errors(path: Path, rel: str, *, is_bundle_root: bool) -> list[str]:
    """Spec §8/§9: index.md frontmatter only for bundle-root okf_version; log.md none."""
    text = path.read_text(encoding="utf-8")
    if not text.startswith("---\n"):
        return []
    fm, _body, fm_errors = _markdown_frontmatter(path)
    if fm_errors:
        return [f"{rel}: {err}" for err in fm_errors]
    if path.name == "log.md":
        return [f"{rel}: reserved log.md must carry no frontmatter"]
    if not is_bundle_root or set(fm) != {"okf_version"}:
        return [f"{rel}: reserved index.md frontmatter is limited to okf_version at the bundle root"]
    return []
```
