  ```python
  from pathlib import Path
  from typing import Any

          "pending": "engine.dashboard.assemble_dashboard (I1 plan H.1)",
      }
  ```

  ```python
  LAYOUT_COLUMNS = 80


  def render_findings(findings: list[dict[str, Any]]) -> list[str]:
      """cli-voice-findings (U2 spec §4): the honesty-card fields verbatim —
      finding/action, argument-for/against only when both sides are present
      (one-sided arguments drop — V2's card grammar), tipped-by, coarse
      certainty. Never a verdict line, never a pre-selected action. One
      function, used by every panel (and by section T's preview screen)."""
      lines: list[str] = []
      for card in findings:
          headline = str(card.get("finding") or card.get("action") or "")
          if headline:
              lines.extend(_fit("  - ", headline))
          if card.get("argument_for") and card.get("argument_against"):
              lines.extend(_fit("    for: ", str(card["argument_for"])))
              lines.extend(_fit("    against: ", str(card["argument_against"])))
          if card.get("what_tipped_it"):
              lines.extend(_fit("    tipped by: ", str(card["what_tipped_it"])))
          if card.get("certainty"):
              lines.extend(_fit("    certainty: ", str(card["certainty"])))
      return lines


  def render_deep(payload: dict[str, Any]) -> str:
      """The deep screen (U2 spec §1/§2): fixed panel order, plain sequential
      text, static per invocation. No curses, no ANSI, no tty branching —
      `memoria cockpit | cat` is byte-identical by construction."""
      if payload.get("resolution") == "ambiguous":
          return _render_ambiguous(payload)
      panels = payload["panels"]
      lines: list[str] = ["memoria cockpit: deep work"]
      lines.extend(_fit("project: ", str(payload.get("project") or "")))
      lines.append("")

      project = panels["project"]
      lines.append(_heading("project", project))
      lines.extend(_fit("  title: ", project["title"]))
      if project["thesis"]:
          lines.extend(_fit("  thesis: ", project["thesis"]))
      lines.append(f"  archived: {'yes' if project['archived'] else 'no'}")
      lines.append("")

      slice_panel = panels["slice"]
      lines.append(_heading("slice", slice_panel))
      lines.append(f"  members: {slice_panel['members']}")
      edges = " ".join(
          f"{kind}={count}" for kind, count in slice_panel["edges_by_type"].items()
      )
      lines.append(f"  edges: {edges or 'none'}")
      lines.append(f"  unresolved outline ids: {slice_panel['missing']}")
      lines.append("")

      draft = panels["draft"]
      lines.append(_heading("draft", draft))
      lines.extend(_fit("  draft path: ", draft["draft_path"]))
      lines.append(f"  draft present: {'yes' if draft['draft_present'] else 'no'}")
      states = " ".join(
          f"{state}={count}" for state, count in draft["evidence_states"].items()
      )
      lines.append(f"  evidence states: {states or 'none'}")
      lines.append(f"  review required: {draft['review_required']}")
      lines.append("")

      grounds = panels["grounds"]
      lines.append(_heading("grounds", grounds))
      lines.append(f"  complete evidence sets: {grounds['complete']}/{grounds['total']}")
      lines.extend(render_findings(grounds["findings"]) or ["  (no gaps or thin claims)"])
      lines.append("")

      trace = panels["trace"]
      lines.append(_heading("recent machine changes", trace))
      lines.extend(_trace_lines(trace))
      lines.append("")

      context = panels["context"]
      lines.append(_heading("context handoff", context))
      lines.extend(_context_lines(context))
      return _buffer(lines)


  def _trace_lines(trace: dict[str, Any]) -> list[str]:
      if "pending" in trace:
          return _fit("  pending: ", str(trace["pending"]))
      lines: list[str] = []
      for event in trace.get("events") or []:
          ref = str(event.get("event_id", ""))
          summary = " ".join(
              str(event[key])
              for key in ("timestamp", "event_type", "output_id")
              if event.get(key)
          )
          lines.extend(_fit(f"  ref {ref}: ", summary or "(no summary fields)"))
      if "total" in trace and "shown" in trace:
          lines.append(f"  showing {trace['shown']} of {trace['total']}")
      if lines:
          lines.append("  refs preview via trace.revert_preview")
          return lines
      return ["  (no machine changes recorded)"]


  def _context_lines(context: dict[str, Any]) -> list[str]:
      if "reserved" in context:
          return _fit("  reserved: ", str(context["reserved"]))
      lines: list[str] = []
      for key in sorted(context.get("bundle") or {}):
          lines.extend(_fit(f"  {key}: ", _scalar(context["bundle"][key])))
      lines.extend(_fit("  invocation: ", str(context["invocation"])))
      return lines


  def _render_ambiguous(payload: dict[str, Any]) -> str:
      projects = payload.get("projects") or []
      lines = ["memoria cockpit: active-project resolution", ""]
      if projects:
          lines.append(f"{len(projects)} active projects; pass --project <path>:")
          for row in projects:
              lines.extend(_fit("  ", f"{row['path']}  ({row['title']})"))
      else:
          lines.append("no active projects (type: project, archived not True)")
          lines.append("pass --project <path> to open one directly")
      return _buffer(lines)


  def _heading(label: str, panel: dict[str, Any]) -> str:
      source = str(panel.get("source_action") or "")
      return f"{label} ({source})" if source else f"{label} (no registry row yet)"


  def _fit(prefix: str, value: str) -> list[str]:
      """80-column layout target; identifiers are never wrapped
      mid-identifier — a token longer than the layout renders whole on its
      own line (keep-test, U2 spec §2)."""
      line = f"{prefix}{value}"
      if len(line) <= LAYOUT_COLUMNS:
          return [line]
      return textwrap.wrap(
          line,
          width=LAYOUT_COLUMNS,
          subsequent_indent=" " * len(prefix),
          break_long_words=False,
          break_on_hyphens=False,
      )


  def _scalar(value: Any) -> str:
      if isinstance(value, str):
          return value
      return json.dumps(value, ensure_ascii=False, sort_keys=True)


  def _buffer(lines: list[str]) -> str:
      return "\n".join(lines).rstrip("\n") + "\n"
  ```

  ```python
  """Cockpit assembly (U2 spec §1): panel builders over U1 registry reads.

  Every builder is a pure read returning one panel dict that carries its
  `source_action` registry id (U2 spec §2). The module holds no state.
  """

  from __future__ import annotations

  from pathlib import Path
  from typing import Any

  from memoria_vault.runtime import state
  from memoria_vault.runtime.policy.paths import normalize_path

  TRACE_PANEL_LIMIT = 10


  def build_trace_panel(
      vault: Path, project_slice: dict[str, Any], *, limit: int = TRACE_PANEL_LIMIT
  ) -> dict[str, Any]:
      """Panel 5 (U2 spec §1): recent machine changes touching the project slice.

      Journal-derived: walks state.read_event_log (ascending event order),
      keeps events whose `output_id` or `target_id` intersects the ids the
      project.slice.read payload exposes, drops PI-actor events (this is the
      *machine* trace), renders newest first, and caps at `limit` while
      reporting the honest total. Each entry carries the ref that the
      trace.revert_preview read (U2 spec §3) consumes.
      """
      vault = Path(vault)
      slice_ids = {
          normalize_path(str(project_slice.get("project_path") or "")),
          normalize_path(str(project_slice.get("outline_path") or "")),
      }
      slice_ids.discard("")
      for member in project_slice.get("members") or []:
          path = member.get("path") if isinstance(member, dict) else None
          if isinstance(path, str) and path:
              slice_ids.add(normalize_path(path))

      matches: list[dict[str, Any]] = []
      for event in state.read_event_log(vault):
          if event.get("actor") == "pi":
              continue
          ref = event.get("output_id") or event.get("target_id")
          if not isinstance(ref, str) or normalize_path(ref) not in slice_ids:
              continue
          matches.append(
              {
                  "ref": normalize_path(ref),
                  "event": str(event.get("event") or event.get("type") or "event"),
                  "operation": str(event.get("operation") or event.get("workflow") or ""),
                  "actor": str(event.get("actor") or ""),
                  "machine": str(event.get("machine") or ""),
                  "timestamp": str(event.get("timestamp") or ""),
              }
          )

      matches.reverse()  # read_event_log is ascending; the panel is newest first
      return {
          "source_action": "cockpit.read",
          "preview_action": "trace.revert_preview",
          "total": len(matches),
          "limit": limit,
          "truncated": len(matches) > limit,
          "entries": matches[:limit],
      }
  ```

  ```python
      """Read-only preview of what ``cascade_rollback`` would touch (U2 spec §3).

      derived-event record, the descendant split the cascade_rollback loop

      per-output materialized commit, the owning explicit operation, and
      whether a whole-workspace backup exists. Mutates nothing.
      """
      vault = Path(vault)
      target = normalize_path(ref)
      event = _latest_derived(vault).get(target)
      if event is None:
          return {
              "computable": False,
              "ref": target,
              "missing": f"derived event record for {target}",
          }

      would_quarantine: list[str] = []
      would_flag: list[str] = []
      would_skip: list[str] = []
      seen: set[str] = set()
      for descendant in trace_downstream(vault, target):
          output_id = descendant.get("output_id")
          if not isinstance(output_id, str):

          elif not (vault / output_id).is_file():

          else:
              would_quarantine.append(output_id)

      record = state.output_record(vault, target)
      backup = local_backup_status(vault)

          "computable": True,
          "ref": target,
          "event": {

              "output_sha256": event.get("output_sha256"),
              "staging_id": event.get("staging_id"),
              "inputs": event.get("inputs") or [],
  ```

  ```python
  def read_context(workspace: Path, *, read_scope: list[str] | None = None) -> dict[str, Any]:
      """context.read (U2 spec §1 panel 6): the situated-context bundle.

      Minimal honest bundle from shipped reads only: active projects (the
      active predicate: `type: project` whose frontmatter `archived` is not
      True), slice counts when exactly one project is active, the open
      attention count, and the effective steering tokens (alphabetical
      top-20 of the same token set discovery relevance consumes) with the
      honest total. No model judgment anywhere in assembly.
      """
      workspace = Path(workspace)
      active: list[dict[str, str]] = []
      for path in iter_markdown(workspace):
          rel = path.relative_to(workspace).as_posix()
          if not _scope_allows(rel, read_scope):
              continue
          frontmatter = read_frontmatter(path)
          if str(frontmatter.get("type") or "") != "project":
              continue
          if frontmatter.get("archived") is True:
              continue
          active.append({"path": rel, "title": str(frontmatter.get("title") or path.stem)})
      active.sort(key=lambda row: row["path"])

      slice_counts = None
      if len(active) == 1:
          project_slice = _read_project_slice(workspace, active[0]["path"])
          slice_counts = {
              "project_path": project_slice["project_path"],
              "members": len(project_slice["members"]),
              "edges": len(project_slice["edges"]),
              "missing": len(project_slice["missing"]),
          }

      open_cards = read_attention(workspace, status="open", read_scope=read_scope)["attention"]
      tokens = sorted(_steering_tokens(workspace))
      return _read_payload(
          context={
              "active_projects": active,
              "slice_counts": slice_counts,
              "attention_open": len(open_cards),
              "steering_tokens": tokens[:20],
              "steering_token_count": len(tokens),
          }
      )
  ```

  ```python
  def test_post_seam_triage_uses_registered_seams(
      vault: Path, monkeypatch: pytest.MonkeyPatch
  ) -> None:
      read_scope = ["projects", "inbox"]
      calls: list[dict[str, object]] = []
      dashboard_calls: list[Path] = []
      real_queue = engine_api.evidence_review_queue
      dashboard_row = actions_by_id()["dashboard.read"]
      assert dashboard_row["engine"] == "read_dashboard"
      real_dashboard = getattr(engine_api, str(dashboard_row["engine"]))

      def observed_queue(
          workspace: Path,
          *,
          batch: int = 10,
          read_scope: list[str] | None = None,
          **filters: object,
      ) -> dict[str, object]:
          calls.append(
              {
                  "workspace": workspace,
                  "batch": batch,
                  "read_scope": read_scope,
                  "filters": filters,
              }
          )
          return real_queue(workspace, batch=batch, read_scope=read_scope, **filters)

      monkeypatch.setattr(engine_api, "evidence_review_queue", observed_queue)

      def observed_dashboard(workspace: Path) -> dict[str, object]:
          dashboard_calls.append(workspace)
          return real_dashboard(workspace)

      monkeypatch.setattr(engine_api, str(dashboard_row["engine"]), observed_dashboard)
      monkeypatch.setattr(
          engine_api,
          "read_evidence_review_view",
          lambda *_args, **_kwargs: pytest.fail("cockpit must not call the evidence-review view"),
          raising=False,
      )
      panels = cockpit.assemble_triage(vault, read_scope=read_scope)

      assert calls == [
          {"workspace": vault, "batch": 0, "read_scope": read_scope, "filters": {}}
      ]
      review = panels["review"]
      assert review["source_action"] == "views.evidence_review"
      assert review["counts"] == {"accept": 5, "defer": 5, "open": 1}
      assert review["open"] == review["counts"]["open"] == 1
      assert review["srd_gaps"] == 1

      assert dashboard_calls == [vault]
      attention_flow = real_dashboard(vault)["dashboard"]["attention_flow"]
      flow = panels["flow"]
      assert flow["source_action"] == "dashboard.read"
      assert flow["open_total"] == attention_flow["open_total"]
      assert flow["inflow"] == sum(attention_flow["inflow_by_day"].values())
      assert flow["drain"] == sum(attention_flow["drain_by_day"].values())

      assert "pending" not in review
      assert "pending" not in flow
      rendered = cockpit.render_triage({"screen": "triage", "panels": panels})
      assert "pending:" not in rendered
  ```
